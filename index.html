Integrasi Langsung Google Spreadsheet

Kirim otomatis setiap data yang disimpan ke Google Sheets melalui Apps Script Web App.

Jika dikosongkan, data tetap tersimpan di browser dan dapat diekspor via CSV atau Salin TSV.

Panduan Singkat Setup Google Sheets:
  1. Buka Google Spreadsheet baru Anda, lalu klik menu Ekstensi > Apps Script.
  2. Hapus kode bawaan dan tempel kode di bawah ini, lalu simpan.
  3. Klik Terapkan (Deploy) > Deployment baru, pilih jenis Aplikasi web.
  4. Atur Yang memiliki akses menjadi Siapa saja (Anyone), lalu salin URL Web App ke kolom di atas.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Waktu Input", "NIK", "Nama Lengkap", "Tempat/Tgl Lahir",
      "Jenis Kelamin", "Alamat", "RT/RW", "Kelurahan/Desa",
      "Kecamatan", "Agama", "Status Perkawinan", "Pekerjaan"
    ]);
  }
  
  sheet.appendRow([
    new Date(),
    "'" + data.nik,
    data.nama,
    data.ttl,
    data.jk,
    data.alamat,
    data.rtrw,
    data.kelurahan,
    data.kecamatan,
    data.agama,
    data.status,
    data.pekerjaan
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
